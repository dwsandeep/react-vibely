
function fibonacci(num){
  let a=0;let b=1;
  let r=[]
  for(let i=0;i<num;i++){
    let n= a+b;
    r.push(a)
    a=b;
    b=n
  }
  return r
}

function reverseStr(str){
  let reversed ='';
  for(let i=str.length-1;i>=0;i--){
    reversed+=str[i]
  }
  return reversed
}

function findDuplicates(arr){
  let obj={}
  let dub=[]
  for(let num of arr){
    if(obj[num]){
      obj[num]++
    }else{
      obj[num]=1
    }
  }
  for(let item in obj){
    if(obj[item]>1){
      dub.push(Number(item))
    }
  }
  return dub
}
